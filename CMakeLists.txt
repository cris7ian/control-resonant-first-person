cmake_minimum_required(VERSION 3.20)
project(ExplorationFirstPerson VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(efp_core STATIC src/core.cpp src/state_snapshot.cpp src/camera_geometry.cpp)
target_include_directories(efp_core PUBLIC src)
if(MSVC)
  target_compile_options(efp_core PRIVATE /W4)
else()
  target_compile_options(efp_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(efp_tests tests/core_tests.cpp)
  target_link_libraries(efp_tests PRIVATE efp_core)
  add_test(NAME core COMMAND efp_tests)
  add_executable(efp_geometry_replay tests/geometry_replay.cpp)
  target_link_libraries(efp_geometry_replay PRIVATE efp_core)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CONTROL Resonant requires a 64-bit build")
  endif()
  add_subdirectory(third_party/minhook)
  add_library(ExplorationFirstPerson SHARED src/native.cpp src/game_adapter.cpp src/state_observer.cpp src/camera_override.cpp)
  target_link_libraries(ExplorationFirstPerson PRIVATE efp_core minhook user32 bcrypt)
  target_compile_definitions(ExplorationFirstPerson PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE EFP_VERSION="${PROJECT_VERSION}")
  set_target_properties(ExplorationFirstPerson PROPERTIES PREFIX "" OUTPUT_NAME "ExplorationFirstPerson")
  if(MINGW)
    target_link_options(ExplorationFirstPerson PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(BUILD_TESTING)
    add_executable(efp_native_smoke tests/native_smoke.cpp)
    if(MINGW)
      target_link_options(efp_native_smoke PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_dependencies(efp_native_smoke ExplorationFirstPerson)
    add_custom_command(TARGET ExplorationFirstPerson POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/smoke"
      COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:ExplorationFirstPerson>" "${CMAKE_BINARY_DIR}/smoke/ExplorationFirstPerson.dll"
    )
    add_test(NAME native_startup COMMAND efp_native_smoke "${CMAKE_BINARY_DIR}/smoke/ExplorationFirstPerson.dll")
    add_executable(efp_camera_hook_smoke tests/camera_hook_smoke.cpp)
    target_link_libraries(efp_camera_hook_smoke PRIVATE efp_core minhook user32)
    target_compile_definitions(efp_camera_hook_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MINGW)
      target_link_options(efp_camera_hook_smoke PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME camera_hook_owned_records COMMAND efp_camera_hook_smoke)
  endif()
endif()
